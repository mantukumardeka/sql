with cte as (
select num,id,id-row_number() over(partition by num order by id )as grp
from numbers
)
select num from cte group by num,grp having count(*) >=3

select distinct num from (
			select 
				num,lag(num,1) over (order by id) as prev1,
				lag(num,2)over(order by id) as prev2
			from numbers
) t
where num=prev1 and num = prev2;

    AVG(sales) OVER (

👉 (A,B) and (B,A) are the same pair
👉 Need to find the pair with maximum total duration
| name1 | name2 | duration |
| ----- | ----- | -------- |
| A     | B     | 15       |
| B     | A     | 10       |
| A     | C     | 5        |
| C     | A     | 7        |
| C     | A     | 30       |
| D     | E     | 12       |
| E     | D     | 8        |

SELECT person1, person2, SUM(duration) AS total_duration
FROM (
    SELECT 
        LEAST(name1, name2) AS person1,
        GREATEST(name1, name2) AS person2,
        duration
    FROM calls
) t
GROUP BY person1, person2
ORDER BY total_duration DESC
LIMIT 1;
