-TOP 5 CITIES WITH HIGHEST RVENUEE IN LAST 30 DAYS

=================Group by + limit============

SELECT city, SUM(revenue) total_revenue
FROM orders
GROUP BY city
ORDER BY total_revenue DESC
LIMIT 5;

WITH cte AS (

    SELECT
        city,
        SUM(revenue) AS total_revenue

    FROM orders

    WHERE order_date >= CURDATE() - INTERVAL 30 DAY

    GROUP BY city
),

====================================================

rnk_cte AS (

    SELECT
        city,
        total_revenue,

        DENSE_RANK() OVER(
            ORDER BY total_revenue DESC
        ) AS rnk

    FROM cte
)

SELECT *
FROM rnk_cte

WHERE rnk <= 5;

-write pyspark code for DAILY SALES FOR CATEGERY using orders,items and products table

============

from pyspark.sql import SparkSession
from pyspark.sql.functions import *

spark = SparkSession.builder.appName("DailyCategorySales").getOrCreate()

# orders table
orders_data = [
    (1, '2025-05-01'),
    (2, '2025-05-01'),
    (3, '2025-05-02')
]

orders_schema = ["order_id", "order_date"]

orders_df = spark.createDataFrame(orders_data, orders_schema)

# items table
items_data = [
    (1, 101, 2),
    (1, 102, 1),
    (2, 101, 1),
    (3, 103, 3)
]

items_schema = ["order_id", "product_id", "quantity"]

items_df = spark.createDataFrame(items_data, items_schema)

# products table
products_data = [
    (101, "Electronics", 500),
    (102, "Clothing", 200),
    (103, "Electronics", 300)
]

products_schema = ["product_id", "category", "price"]

products_df = spark.createDataFrame(products_data, products_schema)

# Daily sales for category
result = orders_df.join(items_df, "order_id") \
    .join(products_df, "product_id") \
    .withColumn("sales_amount", col("quantity") * col("price")) \
    .groupBy("order_date", "category") \
    .agg(sum("sales_amount").alias("total_sales")) \
    .orderBy("order_date", "category")

result.show()
