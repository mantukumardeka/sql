select
   product_id,
   DATE_TRUNC(sale_date, MONTH) AS month,
   SUM(sales_amount) AS total_sal,
   RANK() OVER (ORDER BY
SUM(sales_amount) DESC) AS ranks
FROM sales_table
GROUP BY product_id, month;
